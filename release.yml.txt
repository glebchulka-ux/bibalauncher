name: Build release

on:
  push:
    tags: ["v*"]
  workflow_dispatch:
    inputs:
      version:
        description: "Номер версии, например 1.0.1"
        required: true
        default: "1.0.0"

permissions:
  contents: write

jobs:
  build:
    runs-on: windows-latest
    steps:
      - uses: actions/checkout@v4

      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"

      - name: Версия и адрес обновлений
        shell: bash
        env:
          INPUT_VERSION: ${{ github.event.inputs.version }}
        run: |
          if [ "${{ github.event_name }}" = "workflow_dispatch" ]; then VERSION="$INPUT_VERSION"; else VERSION="${GITHUB_REF_NAME#v}"; fi
          echo "VERSION=$VERSION" >> "$GITHUB_ENV"
          printf 'VERSION = "%s"\nUPDATE_REPO = "%s"\n' "$VERSION" "$GITHUB_REPOSITORY" > build_info.py
          cat build_info.py

      - name: Библиотеки
        run: pip install -r requirements.txt pyinstaller

      - name: Сборка exe
        run: pyinstaller --noconfirm --clean --onefile --noconsole --name BibaLauncher --icon assets/icon.ico --collect-all customtkinter --collect-all minecraft_launcher_lib biba.py

      - name: Контрольная сумма
        shell: bash
        run: |
          cd dist
          sha256sum BibaLauncher.exe | cut -d' ' -f1 > BibaLauncher.exe.sha256
          cat BibaLauncher.exe.sha256

      - name: Релиз
        uses: softprops/action-gh-release@v2
        with:
          tag_name: v${{ env.VERSION }}
          name: Biba Launcher ${{ env.VERSION }}
          generate_release_notes: true
          files: |
            dist/BibaLauncher.exe
            dist/BibaLauncher.exe.sha256
